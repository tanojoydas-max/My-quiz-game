class GameScreen(Screen):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.box = BoxLayout(orientation='vertical', padding=20, spacing=15)
        self.add_widget(self.box)
        self.questions = []
        self.idx = 0
        self.scores = {k: 0 for k in DATA.keys()}
        
        # Pick exactly 2 random questions from each category
        for cat, q_list in DATA.items():
            for q in random.sample(q_list, 2):
                q['cat'] = cat
                self.questions.append(q)
        random.shuffle(self.questions)
        self.show_q()

    def show_q(self):
        self.box.clear_widgets()
        if self.idx >= len(self.questions):
            self.show_results()
            return
            
        q = self.questions[self.idx]
        self.box.add_widget(Label(text=f"Category: {q['cat']} ({self.idx+1}/12)", size_hint_y=0.1))
        self.box.add_widget(Label(text=q['q'], font_size='18sp', size_hint_y=0.3))
        
        opts = list(q['a'])
        random.shuffle(opts)
        for o in opts:
            btn = Button(text=o, size_hint_y=0.2)
            btn.bind(on_press=lambda i, choice=o, correct=q['c'], cat=q['cat']: self.nxt(choice, correct, cat))
            self.box.add_widget(btn)

    def nxt(self, choice, correct, cat):
        if choice == correct:
            self.scores[cat] += 1
        self.idx += 1
        self.show_q()

    def show_results(self):
        self.box.clear_widgets()
        self.box.add_widget(Label(text="🏆 Results 🏆", font_size='24sp', size_hint_y=0.2))
        
        for cat, score in self.scores.items():
            stars = "⭐⭐⭐" if score == 2 else ("⭐" if score == 1 else "❌")
            self.box.add_widget(Label(text=f"{cat}: {score}/2 -> {stars}", font_size='16sp'))
            
        btn = Button(text="Restart", size_hint_y=0.2)
        btn.bind(on_press=self.reset)
        self.box.add_widget(btn)

    def reset(self, instance):
        self.idx = 0
        self.scores = {k: 0 for k in DATA.keys()}
        self.questions = []
        for cat, q_list in DATA.items():
            for q in random.sample(q_list, 2):
                q['cat'] = cat
                self.questions.append(q)
        random.shuffle(self.questions)
        self.show_q()
